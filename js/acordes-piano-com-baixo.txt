/* =========================================================
   EQUIVALÊNCIA ENTRE # E b
   ========================================================= */

const ACORDES_EQUIVALENTES = {
  "A#":"Bb",
  "Bb":"A#",

  "C#":"Db",
  "Db":"C#",

  "D#":"Eb",
  "Eb":"D#",

  "F#":"Gb",
  "Gb":"F#",

  "G#":"Ab",
  "Ab":"G#"
};


/* =========================================================
   BUSCAR DESENHO DO ACORDE
   ========================================================= */


function chordNotes(chord){

  const partes=String(chord||'').split('/');
  const acordeBase=partes[0].trim();
  const baixo=partes[1]?.trim();

  let notas=[];

  if(ACORDES[acordeBase]){
    notas=[...ACORDES[acordeBase]];
  }else{

    const match=acordeBase.match(/^([A-G](?:#|b)?)(.*)$/);

    if(!match){
      return [];
    }

    const root=match[1];
    const suffix=match[2]||'';

    const equivalente=ACORDES_EQUIVALENTES[root];

    if(!equivalente){
      return [];
    }

    notas=[...(ACORDES[equivalente+suffix]||[])];
  }

  // Acrescenta a nota do baixo na primeira oitava
  if(baixo && /^[A-G](?:#|b)?$/.test(baixo)){

    const jaExiste=notas.some(n=>
      noteClass(n.nota)===noteClass(baixo) &&
      Number(n.oitava)===1
    );

    if(!jaExiste){
      notas.push({
        nota:baixo,
        oitava:0
      });
    }
  }

  return notas;
}



/* ========================================================= INICIO
   NÚMERO DA NOTA
   ========================================================= */

function noteClass(note){
  return aliases[note]===undefined
    ? null
    : aliases[note];
}


/* =========================================================
   NOME DA FUNDAMENTAL
   ========================================================= */

function chordRootName(chord){
  const m=String(chord||'').match(/^([A-G](?:#|b)?)/);
  return m?m[1]:'';
}


/* =========================================================
   FECHAR POPUP
   ========================================================= */

function closeChordPopup(){

  if(chordPopupState.el){
    chordPopupState.el.remove();
  }

  chordPopupState.el=null;
  chordPopupState.anchor=null;
}


/* =========================================================
   TECLADINHO
   ========================================================= */


function buildMiniKeyboard(notes){

  const wrap=document.createElement('div');
  wrap.className='chord-mini-piano';

  const whiteNotes=[
    ['C',0],['D',0],['E',0],['F',0],
    ['G',0],['A',0],['B',0],

    ['C',1],['D',1],['E',1],['F',1],
    ['G',1],['A',1],['B',1],

    ['C',2],['D',2],['E',2],['F',2],
    ['G',2],['A',2],['B',2]
  ];

  const keyWidth=100/21;

  const isMarked=(note,octave)=>{
    return notes.some(n=>
      noteClass(n.nota)===noteClass(note) &&
      Number(n.oitava)===octave
    );
  };

  whiteNotes.forEach(([note,octave],i)=>{

    const k=document.createElement('button');
    k.type='button';
    k.className='mini-white';

    k.style.left=(i*keyWidth)+'%';
    k.style.width=keyWidth+'%';

    if(isMarked(note,octave)){
      k.classList.add('marked');
    }

    wrap.appendChild(k);
  });

  const blacks=[
    ['C#',0,0],['D#',0,1],['F#',0,3],
    ['G#',0,4],['A#',0,5],

    ['C#',1,7],['D#',1,8],['F#',1,10],
    ['G#',1,11],['A#',1,12],

    ['C#',2,14],['D#',2,15],['F#',2,17],
    ['G#',2,18],['A#',2,19]
  ];

  blacks.forEach(([note,octave,index])=>{

    const k=document.createElement('button');
    k.type='button';
    k.className='mini-black';

    k.style.left=((index+0.7)*keyWidth)+'%';
    k.style.width=(keyWidth*0.58)+'%';

    if(isMarked(note,octave)){
      k.classList.add('marked');
    }

    wrap.appendChild(k);
  });

  return wrap;
}
